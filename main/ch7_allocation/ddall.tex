Consider now a continuous-time, linear time-invariant model given by
\begin{subequations}
  \label{eq:ddall_sys}
  \begin{align}
    \dot{\bbstate}(t) &= A(\mu)\bbstate(t) + B(\mu)\bbaux(t) + P(\mu)\bbexostate(t), & t\in\R_{\geq 0}, \label{eq:ddall_sys_x}\\
    \bbouto(t) & = C_1(\mu)\bbstate(t) + D_{11}(\mu)\bbaux(t) + D_{12}(\mu)\bbexostate(t), & t\in\R_{\geq 0}, \label{eq:ddall_sys_z0}\\
    \bboutp(t) & = C_2(\mu)\bbstate(t) + D_{21}(\mu)\bbaux(t) + D_{22}(\mu)\bbexostate(t), & t\in\R_{\geq 0}, \label{eq:ddall_sys_z}
  \end{align}
\end{subequations}
where $\bbstate(t)\in\R^\bbstatedim$ is the state vector, $\bbaux(t)\in\R^\bbauxdim$ is the auxiliary control input, $\bbexostate(t)\in\R^\bbexostatedim$ is the reference input, and $\bbouto(t)\in\R^{\bboutodim}$ and $\bboutp(t)\in\R^{\bboutpdim}$ are the regulated and the performance outputs, respectively, at time $t$. The reference input is again taken as the full state of the exosystem \eqref{eq:bbexosys}, \emph{i.e.}, $\bbref = \bbexostate$ and $\bbrefdim = \bbexostatedim$. The exosystem  $\bbexo$ is described by the continuous-time, linear time-invariant model
\begin{align}
  \label{eq:ddall_exo}
  \dot{\bbexostate}(t) &= \exodynmat\bbexostate(t), \quad t\in\R_{\geq 0},
\end{align}
where $\bbexostate(t)\in\R^{\bbexostatedim}$ is the state vector at time $t$, with initial condition $\bbexostate(0) = \bbexostate_0$. The interconnection is depicted in \Cref{fig:bbsys}.

It is assumed that the matrices in \eqref{eq:ddall_sys} depend continuously on an unmeasured vector $\mu$ belonging to a set $\mathcal{M}$ that contains the origin: $\mu = 0$ identifies the nominal case. Although in similar cases studied in the literature (see, \emph{e.g.}, \cite{cocetti_linear_2018}), the set $\mathcal{M}$ is assumed to be fully described and compact, these properties are not necessary for the results that follow. To simplify the notation in the following, the dependence of the matrices in \eqref{eq:ddall_sys} on $\mu$ is omitted unless strictly necessary. The properties of the uncertain model are characterized by the following assumption.

\begin{assumption}
  \label{ass:ddall_plant_exo}
  \eqref{eq:ddall_sys} is fully known for $\mu=0$ and the matrix $A$ in \eqref{eq:ddall_sys_x} is Hurwitz for any $\mu\in\mathcal{M}$.
  The matrix $\exodynmat\in\R^{\bbexostatedim\times\bbexostatedim}$ in \eqref{eq:ddall_exo} is such that $\spec(\exodynmat)\subset\C_{0}$; moreover, its eigenvalues are all distinct and possess \emph{commensurate} angular frequencies.\closestatement
\end{assumption}

As an immediate consequence of \Cref{ass:ddall_plant_exo} and of the ``blanket'' \Cref{ass:bbexoorig,ass:bbexopoisson}, the exosystem \eqref{eq:ddall_exo} is Poisson stable (see \Cref{sec:all_design_ssinv}) and its evolution is periodic of period $\bbperiod$ defined by the least common multiple of the individual periods.

For $s\in\C$,
\begin{subequations}
	\begin{align}
		W_{\bbouto\bbaux}(s)&=C_1(sI-A)^{-1}B+D_{11}, \label{eq:ddall_W_z0_u}\\
		W_{\bboutp\bbaux}(s)&=C_2(sI-A)^{-1}B+D_{21}, \label{eq:ddall_W_zp_u}
	\end{align}
\end{subequations}
denote the transfer matrices of \eqref{eq:ddall_sys} from $\bbaux$ to $\bbouto$ and from $\bbaux$ to $\bboutp$, respectively. Recalling \Cref{ass:bbiss,ass:bboveract}, it is necessary to further characterize the over-actuation property in the uncertain case introduced by \eqref{eq:ddall_sys} to ensure that input allocation remains possible.
\begin{assumption}
  \label{ass:ddall_redundancy}
  For any $\mu\in\mathcal{M}$, the system \eqref{eq:ddall_sys_x}, \eqref{eq:ddall_sys_z0} is such that $\bbauxdim>\bboutodim$, $\rank(B) = \bbauxdim$, and $\rank(W_{\bbouto\bbaux}(s)) = \bboutodim$.\closestatement
\end{assumption}
Finally, recall the definition of the cost functional \eqref{eq:bbcost} as well as the properties of the instantaneous cost function $\ell$. To provide a concise description of the control objective in this scenario,
\begin{equation}
  \label{ddall:eta0}
  \eta_0(t,t_0,\bbstate_0,\bbexostate(\tau),\bbaux(\tau)), \quad \tau\in[t_0, t],
\end{equation}
is the output response for output $\bbouto$ at time $t \geq t_0$ of the plant $\bbsys$ initialized at $\bbstate(t_0) = \bbstate_0$ and driven by the inputs $\bbexostate(\cdot)$ and $\bbaux(\cdot)$. The control problem of interest in this context is formulated as follows.

\begin{problem}
  \label{prb:ddall_main_per}
  Consider the systems \eqref{eq:ddall_sys}-\eqref{eq:ddall_exo} and the steady-state cost functional \eqref{eq:bbcost}. Suppose that \Cref{ass:bbiss,ass:ddall_plant_exo,ass:ddall_redundancy} hold. Find, if possible, a \emph{dynamic control allocator} with input $\allin$ and output $\allout$, to be interconnected to $\bbsys$ and $\bbexo$ according to
  \begin{subequations}
    \label{eq:ddall_all_interc}
    \begin{align}
      \allin &= \bbexostate,\\
      \allout &= \bbaux,
    \end{align}
  \end{subequations}
  such that the following requirements hold.
  \begin{enumerate}[(i)]
    \item \emph{Fixed-time robust output invisibility}: there exists a finite time $t_0 \geq 0$ with the property that, for any $\mu\in\mathcal{M}$, the regulated output $\bbouto$ satisfies
    \begin{equation}
      \label{eq:ddall_output_invis}
      \eta_0(t,t_0,\bbstate_0,\bbexostate(\tau),0) = \eta_0(t,t_0,\bbstate_0,\bbexostate(\tau),\bbaux(\tau)), \quad t \geq t_0,\,\tau\in[t_0, t],\,(\bbstate_0,\bbexostate_0)\in\R^{\bbstatedim}\times\bbexoinitset.
    \end{equation}
    \item \emph{Steady-state periodicity}: for any $\bbexostate_0\in\bbexoinitset$, the internal state of the control allocator is bounded for any time and its unique steady-state response is periodic of period $\bbperiod$.
    \item \emph{Optimality}: the Lagrange cost functional \eqref{eq:bbcost} is minimized with respect to the class of inputs in a \emph{prescribed} finite-dimensional function space.
  \end{enumerate}
  \closestatement
\end{problem}

The construction of a solution to \Cref{prb:ddall_main_per} for any $\mu\in\mathcal{M}$ begins by identifying the data whose knowledge is necessary for the design of the allocator. This knowledge is obtained in the uncertain case by means of algorithms that are described in the following and are based on input-output measurements. Let the spectrum of the matrix $\exodynmat$ in \eqref{eq:ddall_exo} be
\begin{equation}
  \label{eq:ddall_exo_spec}
  \spec(\exodynmat) = \{0, j \omega_1, -j \omega_1, ..., j \omega_{\bar{n}_e}, - j \omega_{\bar{n}_e} \}, \quad \bar{n}_e = \frac{\bbexostatedim-1}{2}.
\end{equation}
Recall \Cref{def:bbreddeg}. The design of the allocator requires the following information.
\begin{definition}[Data for the dynamic control allocator \eqref{eq:ddall_control_all}]
  \label{def:ddall_data}
  The data necessary for the design of the dynamic control allocator \eqref{eq:ddall_control_all} is divided into three items.
  \begin{description}
    \item[K1] The \emph{steady-state map} $\bar{M}_{\bboutp, \bbexostate}$ from $\bbexostate$ to $\bboutp$, \emph{i.e.}, a matrix such that, denoting by $\bbssoutp^\bbexostate(\cdot)$ the part of the steady-state response in $\bboutp$ due to $\bbexostate$,
      \begin{equation}
        \label{eq:ddall_K1mem}
        \bbssoutp^\bbexostate(t) = \bar{M}_{\bboutp, \bbexostate}\bbexostate(t), \quad t\in\R_{\geq 0}.
      \end{equation}
    \item[K2] The \emph{moments} of the transfer matrix $W_{\bboutp\bbaux}(s)$ of \eqref{eq:ddall_sys} at $s\in\{0, j \omega_1,, ..., j \omega_{\bar{n}_e}\}\subset\spec(S)$, namely
      \begin{equation}
        \label{eq:ddall_K2mom}
        W_{\bboutp\bbaux}(0), W_{\bboutp\bbaux}(j \omega_1), \ldots, W_{\bboutp\bbaux}(j \omega_{\bar{n}_e}).
      \end{equation}
    \item[K3] A \emph{polynomial annihilator} for the transfer matrix $W_{\bbouto\bbaux}(s)$ of \eqref{eq:ddall_sys}, namely $W_a(s)\in\mathbb{C}^{\bbauxdim\times m_a}$, with poles with negative real part, such that
      \begin{equation}
        \label{eq:ddall_W_a}
        W_{\bbouto\bbaux}(s)W_a(s)=0, \quad s\in\C\setminus(\poleset(W_{\bbouto\bbaux}(s))\cup\poleset(W_a(s))).
      \end{equation}
  \end{description}\closestatement
\end{definition}

If the data \textbf{K1}, \textbf{K2}, and \textbf{K3} are available, the proposed dynamic allocator can be constructed as follows. Let $(A_a, B_a, C_a)$ be a minimal realization of $W_a(s)$ in \eqref{eq:ddall_W_a}. Define
\begin{subequations}
  \label{eq:ddall_Omega_Lambda}
	\begin{align}
		\Omega & = \diag \left(0, \begin{bmatrix}0 & \omega_{1} \\ - \omega_1 & 0\end{bmatrix},\ldots,\begin{bmatrix}0 & \omega_{\bar{n}_e} \\ - \omega_{\bar{n}_e} & 0\end{bmatrix}\right), \label{eq:ddall_Omega} \\
		\Lambda & = \bbmat{c:cc:c:cc}{c}{1 & 1 & 0 & \cdots & 1 & 0}, \label{eq:ddall_Lambda}
	\end{align}
\end{subequations}
yielding the matrices $(A_{IM}, C_{IM})$:
\begin{subequations}
  \label{eq:ddall_Aim}
  \begin{align}
    A_{IM} & = \Omega \otimes I_{m_a}, \label{eq:ddall_Aim_Aim} \\
    C_{IM} & = \Lambda \otimes I_{m_a}.
  \end{align}
\end{subequations}
Furthermore, using the data \textbf{K2}, define
\begin{equation}
  \label{eq:ddall_Woutpallin}
  W_{\bboutp\allin}(j\omega)=W_{\bboutp\bbaux}(j\omega)W_{a}(j\omega), \quad \omega\in\{0, \omega_1, ..., \omega_{\bar{n}_e}\},
\end{equation}
and build the matrix $\bar{N}_{\mu}$ as
\begin{equation}
  \label{eq:ddall_Nbar_mu}
  \begin{split}
  &\bar{N}_{\mu} =\\
  &\hspace{-1.5pt}\bbmat{c|cc|c|cc}{c}{
    W_{\bboutp\allin}(0)\,
    & \,\re(W_{\bboutp\allin}(j\omega_1))\;
    & \im(W_{\bboutp\allin}(j\omega_1))\,
    & \,\cdots\,
    & \,\re(W_{\bboutp\allin}(j\omega_{\bar{n}_e}))\;
    & \im(W_{\bboutp\allin}(j\omega_{\bar{n}_e}))
  }.
  \end{split}
\end{equation}
Define the functions $\xi(t)$ and $L(\bbexostate(t),x_{IM}(t))$ as
\begin{subequations}
  \label{eq:ddall_G_xi}
	\begin{align}
		\xi(t) & = \bar{M}_{\bboutp, \bbexostate}e^{\exodynmat t}\bbexostate(t) + \bar{N}_{\mu}e^{A_{IM}t}x_{IM}(t), & t\in\R_{\geq 0}, \label{eq:ddall_xi} \\
		L(\bbexostate(t),x_{IM}(t)) & = \left( \int_{0}^{\bbperiod} \nabla \ell(\xi(\sigma))'\bar{N}_{\mu}e^{A_{IM}\sigma} d\sigma \right)', & t\in\R_{\geq 0}. \label{eq:ddall_delta_def}
	\end{align}
\end{subequations}
The proposed dynamic allocator is given by
\begin{subequations}
  \label{eq:ddall_control_all}
	\begin{align}
		\dot{x}_{IM}(t) & = A_{IM}x_{IM}(t) - \gamma L(\bbexostate(t),x_{IM}(t)), & t\in\R_{\geq 0}, \label{eq:ddall_control_all_xim} \\
		\dot{x}_a(t)    & = A_a x_a(t) + B_a C_{IM} x_{IM}(t), & t\in\R_{\geq 0}, \label{eq:ddall_control_all_xa}                  \\
		\allout(t)      & = C_a x_a(t), & t\in\R_{\geq 0}. \label{eq:ddall_control_all_ya}
	\end{align}
\end{subequations}

\begin{remark}
  \label{rem:ddall_structure}
  The equations \eqref{eq:ddall_G_xi}-\eqref{eq:ddall_control_all} allow the designer to directly implement the proposed allocation scheme (more insights on how to do this in software routines as well as on devices such as, \emph{e.g.}, digital microcontrollers, are given in the following). Nonetheless, a deeper understanding of \eqref{eq:ddall_control_all} is necessary to clarify its construction with respect to the architecture introduced in \cite{cristofaro_output_2014}, anticipating the following results. The optimizer unit is given by \eqref{eq:ddall_G_xi} and it embeds, in essence, a gradient descent scheme as shown in the following. The state vector $x_{IM}(t)$ is the state of the steady-state generator at time $t$, whose dynamics are defined by \eqref{eq:ddall_Omega_Lambda}-\eqref{eq:ddall_Aim}. The state vector $x_a(t)$ is the state, at time $t$, of the polynomial annihilator given by $W_a(s)$ and the matrices $(A_a,B_a,C_a)$ are a minimal realization of it.\closeremark
\end{remark}

\begin{remark}
  \label{rem:ddall_Aim}
	The definition of $A_{IM}$ given in \eqref{eq:ddall_Aim_Aim}, together with its eigenvalues selected by \eqref{eq:ddall_Omega}, define the \emph{prescribed} function space mentioned in item (iii) of \Cref{prb:ddall_main_per}: it consists of the functions of time generated as combinations of the modes of \eqref{eq:ddall_exo}, \emph{i.e.}, any function	$v(t)$ such that there exist constants $c_0$, $c_i$, $\phi_i$, $i = 1, \dotsc, \bar{n}_e$, with the property that $v(t) = c_0+\sum_{i = 1}^{\bar{n}_e}c_i \sin(\omega_i t + \phi_i)$.\closeremark
\end{remark}

\begin{remark}
  \label{rem:ddall_galpett}
  In \cite[Problem 1]{galeani_dynamic_2014}, a formulation similar to \Cref{prb:ddall_main_per} is provided, although concerning the only case of $\mu=0$. The results shown therein lead to the development of a dynamic control allocator that is implemented as a hybrid dynamical system, which has also been compared against the solution of the optimal control approach presented in \Cref{sec:all_design_ssinv}. With respect to that work, the solution proposed in \eqref{eq:ddall_control_all} has two main advantages. First, it is a continuous-time system with no hybrid dynamics, which also simplifies its implementation. Secondly, the solution presented in \cite{galeani_dynamic_2014} is based on a gradient descent approach, as is the one hereby proposed (see \eqref{eq:ddall_control_all_xim}), and both these solutions depend on a gradient descent step paramter $\gamma\in\R_{>0}$. What differentiates the proposed approach from the state of the art is that, while in \cite{galeani_dynamic_2014} the step parameter $\gamma$ has to be tuned carefully to ensure convergence due to the hybrid dynamics, here it can be set \emph{arbitrarily large}, thus leading to an arbitrarily fast allocation action.\closeremark
\end{remark}

The equations \eqref{eq:ddall_control_all} allow the designer to directly implement the proposed control allocator. The properties of this solution are clarified in the following. In the case of $\mu=0$, the gathering of the data \textbf{K1}, \textbf{K2}, and \textbf{K3} is straightforward.

\begin{remark}[\textbf{K1}, \textbf{K2}, \textbf{K3} for $\mu=0$]
  \label{rem:ddall_nominalK123}
  If $\mu=0$, \eqref{eq:ddall_sys} is fully known by \Cref{ass:ddall_plant_exo} and the data \textbf{K1}, \textbf{K2}, and \textbf{K3} is computed as follows.
  \begin{description}
    \item[K1] Solve the following:
      \begin{subequations}
        \label{eq:ddall_Mbar_complete}
        \begin{align}
          \Pi_{\Sigma}\exodynmat &= A\Pi_{\Sigma} + P,\label{eq:ddall_Mbar_completea} \\
          \bar{M}_{\bboutp,\bbexostate} &= C_2\Pi_{\Sigma} + D_{22}.\label{eq:ddall_Mbar}
        \end{align}
      \end{subequations}
    \item[K2] $\{W_{\bboutp\bbaux}(0), W_{\bboutp\bbaux}(j\omega_1), \dotsc, W_{\bboutp\bbaux}(j\omega_{\bar{n}_e})\}$ are given by
      \begin{equation}
        W_{\bboutp\bbaux}(s), \quad s\in\{0,j\omega_1,\ldots,j\omega_{\bar{n}_e}\}.
      \end{equation}
    \item[K3] Find $N_a(s)$ whose columns are a minimal polynomial basis for the null space of $W_{\bbouto\bbaux}(s)$. Choose a Hurwitz monic polynomial $d_a(s)$ of degree larger than each element of $N_a(s)$. Define
    \begin{equation}
      \label{eq:ddall_Wa}
      W_a(s) = d_a^{-1}(s)N_a(s),
    \end{equation}
    and choose $(A_a,B_a,C_a,0)$ as any minimal realization of $W_a(s)$.\closeremark
  \end{description}
\end{remark}

The following statement characterizes the properties of \eqref{eq:ddall_control_all} in the case of $\mu=0$.

\begin{lemma}
  \label{lem:ddall_nominal}
  Consider the system \eqref{eq:ddall_sys} with $\mu=0$. Then, the dynamic control allocator \eqref{eq:ddall_control_all} interconnected to \eqref{eq:ddall_sys} as in \eqref{eq:ddall_all_interc} solves \Cref{prb:ddall_main_per} for any $\gamma\in\R_{\geq 0}$ with $t_0=0$.\closestatement
\end{lemma}
\begin{proof}
  \label{pf:ddall_nominal}
  Item (i) of \Cref{prb:ddall_main_per} with $\mu=0$ follows immediately by knowledge of \textbf{K3} computed as in \Cref{rem:ddall_nominalK123}; this guarantees full output invisibility of the allocator unit as requested. Therefore, the optimality requirement remains to be addressed. To this end, it must be shown that the behavior of the proposed allocator \eqref{eq:ddall_control_all} leads to the minimization of the cost functional \eqref{eq:bbcost}. This is achieved by showing that \eqref{eq:ddall_control_all_xim}, \eqref{eq:ddall_delta_def} define a gradient descent algorithm for the cost functional \eqref{eq:bbcost}. Define the \emph{moving-horizon cost} over a period as
  \begin{equation}
    \label{eq:ddall_cost_moving}
    J_{t} = \int_{t}^{t+\bbperiod}\ell(\bbssoutp(\sigma))\,d\sigma, \quad t\in\R_{\geq 0},
  \end{equation}
  and consider \eqref{eq:ddall_control_all_xim} with an input $u_{IM}(t)$ to be determined, \emph{i.e.},
  \begin{align}
    \label{eq:ddall_control_all_xim_mod}
    \dot{x}_{IM}(t) & = A_{IM}x_{IM}(t) + u_{IM}(t), \quad t\in\R_{\geq 0}.
  \end{align}
  To evaluate the effect of a change in the cost $J$ due to a change in $x_{IM}$, consider the following selections of $u_{IM}$.
  \begin{subequations}
    \label{eq:ddall_u_IM}
  	\begin{align}
  		u^\delta_{IM}(\sigma) &=
  		\begin{cases}
  			\delta, & \text{ if } \sigma \in [t,t+\Delta],\,\Delta>0,\,t\in\R_{\geq 0}, \\
  			0,      & \text{ otherwise. }
  		\end{cases}\label{eq:ddall_u_IM_delta} \\
  		u^\circ_{IM}(\sigma) &= 0, \quad \forall \sigma\geq t,\,t\in\R_{\geq 0}.\label{eq:ddall_u_IM_circ}
  	\end{align}
  \end{subequations}
  Denote the respective evolutions of $\bbssoutp$, $x_{IM}$, and $J_t$ by the superscripts $\delta$ or $\circ$, \emph{e.g.}, $\bbssoutp^\delta$ and $\bbssoutp^\circ$. Integrating \eqref{eq:ddall_control_all_xim_mod}, \eqref{eq:ddall_u_IM} from the
  common initial condition $x_{IM}^{\circ}(t)$ yields
  \begin{subequations}
    \begin{align}
      x_{IM}^{\circ}(t+\Delta) & = e^{A_{IM}\Delta}x_{IM}^{\circ}(t),\\
      \begin{split}
        x_{IM}^\delta(t+\Delta)  & = e^{A_{IM}\Delta}x_{IM}^{\circ}(t) + \int_{0}^{\Delta} e^{A_{IM}\tau}u_{IM}(t+\Delta-\tau)\,d\tau \\
        &= x_{IM}^{\circ}(t+\Delta) + \int_{0}^{\Delta}e^{A_{IM}\tau}\,d\tau\delta = x_{IM}^{\circ}(t+\Delta)+\Delta\bar{B}\delta.
      \end{split}
    \end{align}
  \end{subequations}
  The matrix $\bar{B}$ is such that
  \begin{align}
    \label{eq:ddall_norm_Bbar}
    \|\bar{B}\| & \leq1, \qquad \lim_{\Delta\rightarrow0^{+}}\bar{B}=I,
  \end{align}
  which follow from the fact that $\|e^{A_{IM}\tau}\|=1$ (since $(e^{A_{IM}\tau})'e^{A_{IM}\tau}=I$) and from integration of the series expansion of $e^{A_{IM}\tau}$, respectively. Since $u_{IM}^\circ$ and $u_{IM}^\delta$ are zero after $t+\Delta$, it holds that
  \begin{equation}
    x_{IM}^{\delta}(\tau)-x_{IM}^{\circ}(\tau) = e^{A_{IM}(\tau-(t+\Delta))}\Delta\bar{B}\delta, \quad \tau\geq t+\Delta,
  \end{equation}
  and the corresponding steady-state responses of $\bboutp$ are
  \begin{subequations}
    \begin{align}
      \bbssoutp^{\circ}(\tau)  & = \bar{M}_{\bboutp,\bbexostate}\bbexostate(\tau) + \bar{N}_\mu x_{IM}^{\circ}(\tau),\\
      \bbssoutp^{\delta}(\tau) & = \bar{M}_{\bboutp,\bbexostate}\bbexostate(\tau) + \bar{N}_\mu x_{IM}^{\delta}(\tau),\\
        \hat{\tilde{\eta}}_p(\tau)  & = \bbssoutp^{\delta}(\tau)-\bbssoutp^{\circ}(\tau) = \bar{N}_\mu e^{A_{IM}(\tau-(t+\Delta))}\Delta\bar{B}\delta.\label{eq:ddall_ztilde}
    \end{align}
  \end{subequations}
  If $\Delta$ is sufficiently small, the difference between the two cost integrand functions along the two solutions satisfy
  \begin{align}
    \begin{split}
	    \ell(\bbssoutp^{\delta}(\tau)) - \ell(\bbssoutp^{\circ}(\tau)) & = \left.\nabla \ell\right|_{\bbssoutp^{\circ}(\tau)} \hat{\tilde{\eta}}_p(\tau) + o(\|\hat{\tilde{\eta}}_p(\tau)\|)\\
	    & = \left.\nabla \ell\right|_{\bbssoutp^{\circ}(\tau)}\bar{N}_\mu e^{A_{IM}(\tau-(t+\Delta))}\Delta\bar{B}\delta + o(\|\Delta\|),
    \end{split}
  \end{align}
  where the equality $o(\|\hat{\tilde{\eta}}_p(\tau)\|)=o(\|\Delta\|)$ follows from \eqref{eq:ddall_ztilde}, \eqref{eq:ddall_norm_Bbar}, implying in turn that
  \begin{equation}
    \label{eq:ddall_mario}
	  \|\bar{N}_\mu e^{A_{IM}(\tau-(t+\Delta))}\Delta\bar{B}\delta\|\leq
	  \|\bar{N}_\mu\| \|\Delta\| \|\delta\|,
  \end{equation}
  provided that $||\delta||$ is bounded by a quantity independent of $\Delta$, which is shown in the following.
  Periodicity of $\bbssoutp^{\circ}$ yields
  \begin{equation}
    J^\circ_{t} = J^\circ_{t+\Delta}, \quad t\in\R_{\geq 0},
  \end{equation}
  hence, for small $\Delta$:
  \begin{align}
    \begin{split}
	    J^\delta_{t+\Delta} - J^\circ_{t} & = J^\delta_{t+\Delta} - J^\circ_{t+\Delta} \\
                                        & =	\int_{t+\Delta}^{t+\Delta+\bbperiod}\left(\ell(\bbssoutp^{\delta}(\tau))-\ell(\bbssoutp^{\circ}(\tau))\right)\,d\tau\\
	                                      & = \int_{t+\Delta}^{t+\Delta+\bbperiod}\left(\left.\nabla \ell\right|_{\bbssoutp^{\circ}(\tau)}\hat{\tilde{\eta}}_p(\tau)+o(\|\Delta\|)\right)\,d\tau\\
                                        & = \int_{t+\Delta}^{t+\Delta+\bbperiod}\left.\nabla \ell\right|_{\bbssoutp^{\circ}(\tau)}\hat{\tilde{\eta}}_p(\tau)\,d\tau+o(\|\Delta\|).
    \end{split}
  \end{align}
  Hence, recalling that $\displaystyle \lim_{\Delta \rightarrow0^{+}}\frac{1}{\Delta}o(\|\Delta\|)=0$ by definition of $o(\cdot)$, and \eqref{eq:ddall_ztilde}, it follows that
  \begin{align}
    \begin{split}
	    \dot{J}_t & = \lim_{\Delta \rightarrow 0^{+}} \dfrac{J^\delta_{t+\Delta}-J^\circ_{t}}{\Delta}\\
	              & = \lim_{\Delta \rightarrow 0^{+}} \frac{1}{\Delta} \int_{t+\Delta}^{t+\Delta+\bbperiod}\left.\nabla \ell\right|_{\bbssoutp^{\circ}(\tau)}\hat{\tilde{\eta}}_p(\tau)\,d\tau\\
	              & = \int_{t}^{t+\bbperiod} \left.\nabla \ell\right|_{\bbssoutp^{\circ}(\tau)}\bar{N}_\mu e^{A_{IM}(\tau-t)}\,d\tau\delta\\
	              & = F(t) \delta, \quad t\in\R_{\geq 0}.
    \end{split}
  \end{align}
  Selecting $\delta = -\gamma F(t)$ for any $\gamma\in\R_{\geq 0}$ yields
  \begin{equation}
    \label{eq:ddall_Jdot_gamma}
    \dot{J}_t = -\gamma\|F(t)\|^2, \quad t\in\R_{\geq 0}.
  \end{equation}
  Note that $F(t)$ only depends on the nominal $\bbssoutp^{\circ}$ trajectory and is independent of $\Delta$, as required after \eqref{eq:ddall_mario}.
	The change of the variable $\sigma = \tau - t$ in the definition of $F(t)$ shows that	$F(t)' = L(\bbexostate(t), x_{IM}(t))$ with the latter defined in \eqref{eq:ddall_G_xi}.
	The latter equation	\eqref{eq:ddall_Jdot_gamma}, in turn, together with boundedness of $\bar{M}_{\bboutp,\bbexostate}\bbexostate(t)$ for any time as in \eqref{eq:ddall_K1mem}, implies that $x_{IM}$ in \eqref{eq:ddall_control_all_xim} is bounded and that, at steady-state, it is described by the differential equation $\dot{\tilde{x}}_{IM}(t) = A_{IM}\tilde{x}_{IM}(t)$, the solution of which is, by construction, periodic of period $\bbperiod$.
  Finally, the annihilator is asymptotically stable since the matrix $A_a$ in \eqref{eq:ddall_control_all_xa} is Hurwitz; hence, the condition on $\dot{J}_{t}$ implies	that items (ii) and (iii) of \Cref{prb:ddall_main_per} hold, concluding the proof.
\end{proof}

\begin{remark}
  \label{rem:ddall_nablaG_comp}
  In practice, the dynamic control allocation scheme \eqref{eq:ddall_control_all} can be implemented by approximating the integral in \eqref{eq:ddall_delta_def} with a finite sum. Given a sufficiently small sampling time $\tau_s\in\R_{>0}$, $\tau_s = \frac{\bbperiod}{K}$, $K\in\N$, define $\bar{\exodynmat} = e^{\exodynmat\tau_s}$ and $\bar{A}_{IM} = e^{A_{IM}\tau_s}$. Then, the gradient of the cost functional can be approximated as
  \begin{subequations}
    \label{eq:ddall_approx_G}
    \begin{align}
      \xi_k & = \bar{M}_{\bboutp,\bbexostate}\bar{\exodynmat}^k \bbexostate(k\tau_s) + \bar{N}_{\mu}\bar{A}_{IM}^k x_{IM}(k\tau_s),\label{eq:ddall_approx_G_xi}\\
      L(\bbexostate(t), x_{IM}(t))' & \approx \sum_{k = 1}^{K}\, \nabla \ell(\xi_k)' \cdot \bar{N}_{\mu}\bar{A}_{IM}^k, \quad t\in\R_{\geq 0}, \label{eq:ddall_approx_G_G}
    \end{align}
  \end{subequations}
  which is a nonlinear function of the measurements of $x_{IM}(t)$ and $\bbexostate(t)$. The core difficulties in the implementation of \eqref{eq:ddall_control_all} arise in the nonlinear terms of \eqref{eq:ddall_approx_G} and fall within the wide spectrum of problems concerning function approximation with digital devices. In this specific case, assuming that the quantities $\bbexostate(t)$ and $x_{IM}(t)$ remain constant during sampling intervals, a digital device with a sampling time equal to $\tau_s$ in \eqref{eq:ddall_approx_G} can be used to implement the control allocator \eqref{eq:ddall_control_all}.\closeremark
\end{remark}

The following result reveals that the structure of the control allocator \eqref{eq:ddall_control_all} is simplified significantly when the cost funcional \eqref{eq:bbcost} is quadratic: the correction term \eqref{eq:ddall_delta_def}-\eqref{eq:ddall_xi} becomes a \emph{static linear feedback} from the state vector in \eqref{eq:ddall_control_all_xim} and a \emph{feedforward} from the state of the exosystem \eqref{eq:ddall_exo}.

\begin{proposition}
  \label{prop:ddall_delay_quadratic}
  Consider the control allocator \eqref{eq:ddall_control_all} and suppose that $\ell(\bbssoutp(t)) = \frac{1}{2}\|\bbssoutp(t)\|^{2}_R$, with $R = R' \succ 0$. Then, $L(\bbexostate(t),x_{IM}(t))$ in \eqref{eq:ddall_xi}-\eqref{eq:ddall_delta_def} becomes
	\begin{equation}
    \label{eq:ddall_delta_quadratic}
		L(\bbexostate(t),x_{IM}(t)) = H_1 x_{IM}(t) +	H_2 \bbexostate(t), \quad t\in\R_{\geq 0},
  \end{equation}
  with
	\begin{subequations}
    \label{eq:ddall_H}
		\begin{align}
      H_1 = & \int_0^{\bbperiod} e^{A_{IM}' \sigma} \bar{N}_{\mu}' R \bar{N}_{\mu}e^{A_{IM} \sigma}\,d\sigma, \\
      H_2 = & \int_0^{\bbperiod} e^{A_{IM}' \sigma} \bar{N}_{\mu}' R \bar{M}_{\bboutp,\bbexostate}e^{\exodynmat\sigma}\,d\sigma.
    \end{align}
	\end{subequations}
  Hence, for any $\gamma\in\R_{> 0}$, \eqref{eq:ddall_control_all_xim} becomes
  \begin{equation}
    \label{eq:ddall_ss_opt_quadratic}
    \dot{x}_{IM}(t) =	(A_{IM}-\gamma H_1) x_{IM}(t) - \gamma H_2 \bbexostate(t), \quad t\in\R_{\geq 0}.
  \end{equation}
	Moreover, if the pair $(A_{IM},R^{\frac{1}{2}}\bar{N}_{\mu})$ is observable, then the matrix $(A_{IM}-\gamma H_1)$ is Hurwitz for any $\gamma\in\R_{> 0}$.\closestatement
\end{proposition}
\begin{proof}
  In this case, the gradient of the cost function $\ell$ in \eqref{eq:bbcost} is linear:
  \begin{equation}
    \nabla\ell(\bbssoutp(t)) = R\bbssoutp(t), \quad t\in\R_{\geq 0}.
  \end{equation}
  Applying standard properties of integrals, \eqref{eq:ddall_delta_def} reduces to
  \begin{equation}
    \begin{split}
      L(\bbexostate(t),x_{IM}(t)) & = \int_{0}^{\bbperiod}  e^{A_{IM}'\sigma}\bar{N}_{\mu}'R\bar{N}_{\mu}e^{A_{IM}\sigma}\,d\sigma x_{IM} + \int_{0}^{\bbperiod} e^{A_{IM}'\sigma}\bar{N}_{\mu}R\bar{M}_{\bboutp,\bbexostate}e^{S\sigma}\,d\sigma \bbexostate(t)\\
                                  & = H_1 x_{IM}(t) + H_2 \bbexostate(t), \quad t\in\R_{\geq 0}.
    \end{split}
  \end{equation}
  The last claim of \Cref{prop:ddall_delay_quadratic} follows from two facts: first, $H_1$ is the observability Gramian matrix of the pair $(A_{IM},R^{\frac{1}{2}}\bar{N}_{\mu})$, and secondly, the matrix $A_{IM}$ is Hurwitz by construction.
\end{proof}

\begin{remark}
  \label{rem:ddall_implementation_quadratic}
  In practice, the matrices $H_1$ and $H_2$ in \eqref{eq:ddall_H} can be computed offline.
  Define
  \begin{subequations}
    \begin{align}
      A^1_{12} &= \bar{N}_{\mu}' R \bar{N}_{\mu}, \\
      A^1_{22} &= A_{IM}, \\
      A^2_{12} &= \bar{N}_{\mu}' R \bar{M}_{\bboutp,\bbexostate}, \\
      A^2_{22} &= \exodynmat,
    \end{align}
  \end{subequations}
  and compute the matrix exponentials
  \begin{equation}
    H_i = \left[\begin{array}{cc} I & 0 \end{array}\right] \exp	\left(\bbm{-A_{IM}' & A_{12}^{i} \\ 0 & A_{22}^{i}}\bbperiod\right)	\left[\begin{array}{c} 0 \\ I \end{array}\right], \quad i\in\{1,2\}.
  \end{equation}
  Furthermore, denoting by $x^\star_{IM}(t)$ the optimal steady-state evolution of $x_{IM}(t)$, the fact that the gradient $L(\bbexostate(t),x^\star_{IM}(t))$ is equal to zero for any $t\in\R_{\geq 0}$ yields
  \begin{equation}
    \label{eq:ddall_xim_opt_quadratic}
    x^\star_{IM}(t) = -H^{-1}_1H_2\bbexostate(t), \quad t\in\R_{\geq 0}.
  \end{equation}
  The first consequence of \eqref{eq:ddall_xim_opt_quadratic} is that the correct initialization for the steady-state generator is $x_{IM}(0) = -H^{-1}_1H_2\bbexostate(0)$. Secondly, one may \emph{bypass} the steady-state generator entirely by redefining the control allocator \eqref{eq:ddall_control_all} as
  \begin{subequations}
    \label{eq:ddall_control_all_quadratic}
    \begin{align}
      \dot{x}_a(t) & = A_ax_a(t) - B_aC_{IM}H_1^{-1}H_2 \bbexostate(t), & t\in\R_{\geq 0},\\
		  y_a(t)       & = C_a x_a(t), & t\in\R_{\geq 0}.
    \end{align}
  \end{subequations}
  \closeremark
\end{remark}

\begin{remark}
  The observability property in the last sentence of \Cref{prop:ddall_delay_quadratic} is always satisfied if the columns of the numerator of the annihilator $W_a(s)$, namely $N_{a}(s)$, are chosen as a minimal basis,	as it can be inferred from \cite[Thm. 6.5.10]{kailath_linear_1980}.\closeremark
\end{remark}

Dealing with the general case of $\mu\in\mathcal{M}$ requires data-driven algorithms for the estimation of the data in \Cref{def:ddall_data}. To this end, the steady-state behavior of the uncertain system \eqref{eq:ddall_sys}-\eqref{eq:ddall_exo} must first be estimated. The following results and algorithms provide a solution to this problem based on the concept of \emph{moment matching} for a linear system; the basic definitions and results required in the following are given in \Cref{app:moments}. Starting from \textbf{K1}, the first item to be estimated is the steady-state map $\bar{M}_{\bboutp,\bbexostate}$. Usually, this implies disconnecting the exosystem \eqref{eq:ddall_sys} from the system \eqref{eq:ddall_sys}; in the current context this procedure is not permitted, since one of the main goals of the allocation approach is to not alter or reconfigure in any way the preexisting control system. Thus, a data-driven algorithm for the online estimation of $\bar{M}_{\bboutp,\bbexostate}$ with $\bbaux=0$ is proposed. Since it is based on sampled measurements of $\bbssoutp(t)$ and $\bbexostate(t)$, the first step consists in defining the characteristics of the sampling procedure. Let $\tau_s\in\R_{>0}$ be the sampling time, characterized by the following statements.
\begin{definition}[Non-pathological sampling time]
  \label{def:ddall_nonpatho}
  The sampling time $\tau_s>0$ is \emph{non-pathological} if, for any $\lambda_1,\lambda_2\in\spec(\exodynmat)\cup\spec(A)$, there is no $h\in\Z$ such that
  \begin{equation}
    \tau_s(\lambda_1-\lambda_2)=h2\pi j.
  \end{equation}
  \closestatement
\end{definition}
\begin{assumption}
  \label{ass:ddall_nonpatho}
  $\tau_s$ is non-pathological.\closestatement
\end{assumption}
Let $\exodynmat_d=e^{\exodynmat\tau_s}$, $A_d=e^{A\tau_s}$. \Cref{ass:ddall_nonpatho} implies that all distinct eigenvalues of $\exodynmat$ and $A$ are mapped to distinct eigenvalues of $\exodynmat_d$ and $A_d$; moreover all structural properties (\emph{e.g.}, observability and reachability) of \eqref{eq:ddall_sys}-\eqref{eq:ddall_exo} are preserved in their sampled data equivalents. More details about this topic are found in \cite[Sec. 3.2]{chen_optimal_1995}.

\begin{algorithm}[h!]
  \caption{Data-driven pseudo-steady-state output map estimation.}
  \label{alg:ddall_algo1}
  \begin{algorithmic}[1]
    % NOTATION OVERRIDE: \eta_{p,k}
    \Require $\tau_s$, $K\in\{a\in\N : a \geq \bbstatedim + 1 + \bboutpdim\bbexostatedim\}$, $\{\bbexostate_k\}_{k=0}^{K-1}$, $\{\eta_{p,k}\}_{k=0}^{K+\bbstatedim-1}$.
    \Ensure $\bar{M}_{\bboutp,\bbexostate}$.

    \State Define $T(\bboutp,\bbexostate)$ as:
    \Statex
    \begin{align}
      \label{eq:ddall_Toeplitz}
      T(\bboutp,\bbexostate) = \bbm{
      \bboutp(n\tau_s)       & \ldots & \bboutp(0)           & -\bbexostate(0)'\otimes I_{\bboutpdim} \\
      \bboutp((n+1)\tau_s)   & \ldots & \bboutp(\tau_s)      & -\bbexostate(\tau_s)'\otimes I_{\bboutpdim} \\
      \vdots                 & \ldots & \vdots               & \vdots \\
      \bboutp((n+K-1)\tau_s) & \ldots & \bboutp((K-1)\tau_s) & -\bbexostate((K-1)\tau_s)'\otimes I_{\bboutpdim}
      }.
    \end{align}

    \State Find the smallest $\nu\in\N$, $1\leq \nu\leq n$, for which there exists $v\in\R^{n+1+\bboutpdim n_e}$ satisfying:
    \Statex
    \begin{align}
      \label{eq:ddall_aM_hat}
      T(\bboutp,\bbexostate)v=0,
    \end{align}
    \Statex with $v$ such that
    \Statex
    \begin{align}
      \label{eq:ddall_structure_v}
      v'=\bbm{0_{1\times (n-\nu)} & 1 &\hat{a}_{\nu-1} & \hat{a}_{\nu-2} & \cdots & \hat{a}_{0} & \hat{M}'}.
    \end{align}

    \State Define $\tilde{M}\in\R^{\bboutpdim\times n_e}$ such that $\ve{\tilde{M}}=\hat{M}$.

    \State Let $\hat{a}_\nu=1$ and compute $\bar{M}_{\bboutp,\bbexostate}$ as:
    \Statex
    \begin{align}
      \label{eq:ddall_Minv}
      \bar{M}_{\bboutp,\bbexostate}=\tilde{M}\left(\sum_{h=0}^{\nu}\hat{a}_h S_d^h \right)^{-1}.
    \end{align}
  \end{algorithmic}
\end{algorithm}

\begin{remark}
  \Cref{alg:ddall_algo1} holds onto \Cref{ass:ddall_plant_exo,ass:ddall_nonpatho}. Moreover, here it is defined for data sampled at times $k=0,1,\dotsc,K-1$ but it can be applied (with obvious changes) to any sequence of $K$ data samples, \emph{i.e.}, for $k=\bar{k},\dotsc,\bar{k}+K-1$ for an arbitrary $\bar{k}\in\Z$.\closeremark
\end{remark}

\begin{proposition}
  \label{prop:ddall_est_mom}
	Let \Cref{ass:ddall_plant_exo,ass:ddall_nonpatho} hold. Fix $\bbexostate_0\in\bbexoinitset$. Then, the following statements hold.
	\begin{enumerate}
		\item[i.] \Cref{alg:ddall_algo1} is correct and yields a matrix $\bar{M}_{\bboutp,\bbexostate}$.
		\item[ii.] $\bar{M}_{\bboutp,\bbexostate}$ satisfies $\bbssoutp(t)=\bar{M}_{\bboutp,\bbexostate}\bbexostate(t)$ for all $t\in\R_{\geq 0}$.
		\item[iii.] If $\bbexostate_0$ excites all natural modes of \eqref{eq:ddall_exo},	then $\bar{M}_{\bboutp,\bbexostate}$ is equal to the unique solution of \eqref{eq:ddall_Mbar}.\closestatement
	\end{enumerate}
\end{proposition}
Note that statement (ii) of \Cref{prop:ddall_est_mom} highlights that the matrix $\bar{M}_{\bboutp,\bbexostate}$ computed by \Cref{alg:ddall_algo1} always ensures a correct prediction of the steady-state evolution \emph{for the trajectory used in the computation}, even if such matrix does not coincide with the unique solution of \eqref{eq:ddall_Mbar}. Further conditions, concerning the quantity of information that the data provides, are required to ensure the conincidence of the two matrices; this is specified in statement (ii) of \Cref{prop:ddall_est_mom}. Since the algorithm is intended to be used for data-driven, online estimation, the trajectory used in the computation (arising from the fixed $\bbexostate_0$) is exactly the one of interest for correctly controlling the plant, hence statement (ii) is the one actually useful in practice. The issue at hand is the same as the one in classic adaptive regulation, where the control objective is achieved even without fully identifying the parameters of the plant. Statement (iii) is given to complete the analysis and provide a full understanding of the issues at play.
\begin{proof}
  To prove item (i) of \Cref{prop:ddall_est_mom}, notice that all steps in \Cref{alg:ddall_algo1} can be performed, provided that \eqref{eq:ddall_aM_hat} is solvable and the matrix inverse in \eqref{eq:ddall_Minv} exists; both facts are shown in the following.
  In the following, given a signal $x(t)$, $x^{\circ}(t)$ denotes its transient response, provided the latter exists.
  For all $k\in\N$, the following holds:
  \begin{equation}
    \label{eq:ddall_mista}
    \bboutp(k\tau_s) = \bboutp^{\circ}(k\tau_s) + \bbssoutp(k\tau_s) = C_2\bbstate^{\circ}(k\tau_s) + \bar{M}_{\bboutp,\bbexostate}\bbexostate(k\tau_s).
  \end{equation}
  Recall \cite[App. A.8]{kailath_linear_1980} that a matrix satisfies the equation defined by its \emph{minimal polynomial} (\emph{i.e.}, the smallest degree polynomial which enjoys this property), and that the minimal polynomial has the same roots, with lower or equal multiplicity, of the \emph{characteristic polynomial}.
  Define $\hat{a}_\nu=1$ and let the minimal (monic) polynomial of $A_d$ be:
  \begin{equation}
    \rho_{A_d}(z) = \sum_{h=0}^{\nu}\hat{a}_h z^h, \quad z\in\C.
  \end{equation}
  Since $\rho_{A_d}(A_d)=0$ by definition and
  \begin{subequations}
    \begin{align}
      \bbexostate((k+h)\tau_s) &= \exodynmat_d^h \bbexostate(k\tau_s), & k\in\Z_{\geq 0}, \\
      \bbstate^{\circ}((k+h)\tau_s) &= A_d^h \bbstate^{\circ}(k\tau_s), & k\in\Z_{\geq 0},
    \end{align}
  \end{subequations}
  it is possible to define a linear combination of the measurements \eqref{eq:ddall_mista} that is independent of $\bbstate^{\circ}_0=\bbstate^{\circ}(0)$ as
  \begin{equation}
    \label{eq:ddall_menomista}
    \begin{split}
      \sum_{h=0}^{\nu} \hat{a}_h \bboutp((k+h)\tau_s) &= \sum_{h=0}^{\nu} \hat{a}_h C_2 \bbstate^{\circ}((k+h)\tau_s) + \sum_{h=0}^{\nu} \hat{a}_h \bar{M}_{\bboutp,\bbexostate}\bbexostate((k+h)\tau_s) \\
                                                      &= C_2 \rho_{A_d}(A_d) \bbstate^{\circ}(k\tau_s) + \bar{M}_{\bboutp,\bbexostate} \rho_{A_d}(\exodynmat_d)\bbexostate(k\tau_s) \\
                                                      &= \bar{M}_{\bboutp,\bbexostate} \rho_{A_d}(\exodynmat_d)\bbexostate(k\tau_s).
    \end{split}
  \end{equation}
  Defining
  \begin{subequations}
    \begin{align}
      H(k\tau_s) &= \bbm{\bboutp((k+\nu)\tau_s) & \bboutp((k+\nu-1)\tau_s) & \cdots & \bboutp((k+1)\tau_s) & \bboutp(k\tau_s)}, \\
      \hat{a}    &= \bbm{\hat{a}_{\nu} & \hat{a}_{\nu-1} & \cdots & \hat{a}_1 & \hat{a}_0}, \\
      \tilde{M}  &= \bar{M}_{\bboutp,\bbexostate}\rho_{A_d}(\exodynmat_d),
    \end{align}
  \end{subequations}
  yields
  \begin{align}
    \label{eq:ddall_menomista2}
    H(k\tau_s)\hat{a}=\tilde{M}\bbstate(k\tau_s).
  \end{align}
  In turn, \eqref{eq:ddall_menomista2} can be vectorized \cite[Sec. 2.5]{zhou_robust_1996} introducing $\hat{M}=\ve{\tilde{M}}$, yielding:
  \begin{align}
    \label{eq:ddall_menomista3}
    H(k\tau_s)\hat{a}=(\bbexostate(k\tau_s)'\otimes I_{\bboutpdim})\hat{M},
  \end{align}
  and collecting \eqref{eq:ddall_menomista3} for $k=0,1,\ldots,K-1$ yields \eqref{eq:ddall_aM_hat}.

  Since the degree of the minimal polynomial of a matrix never exceeds the degree of its characteristic polynomial, the above reasoning proves that a vector $v\in\R^{n+1+\bboutpdim n_e}$ as required at line 2 of \Cref{alg:ddall_algo1} can be found.

  As for the invertibility of the matrix at line 4, note that this matrix is exactly $\rho_{A_d}(\exodynmat_d)$. Since $\exodynmat$ is diagonalizable, there exists an invertible matrix $T_I$ such that $S=T_I\Lambda_\exodynmat T_I^{-1}$, with $\Lambda_\exodynmat$ diagonal. Since $\tau_s$ is non-pathological, $\exodynmat_d$ is diagonalizable by using the same $T_I$ as $\exodynmat$, yielding $\exodynmat_d=T_I\Lambda_{\exodynmat_d} T_I^{-1}$ with $\Lambda_{\exodynmat_d}$ diagonal. By rewriting
  \begin{equation}
    \rho_{A_d}(\exodynmat_d) = \rho_{A_d}(T_I\Lambda_{\exodynmat_d}	T_I^{-1})=T_I\rho_{A_d}(\Lambda_{\exodynmat_d})T_I^{-1},
  \end{equation}
  and considering that $\rho_{A_d}(\Lambda_{\exodynmat_d})$ is diagonal, it is clear that $\rho_{A_d}(\exodynmat_d)$ is not invertible if and only if one of the elements on its diagonal is zero; but this would imply that $\exodynmat_d$ and $A_d$ have at least one eigenvalue in common, which is impossible since by hypothesis $\tau_s$ is non-pathological and $\spec(\exodynmat)\cap\spec(A)=\emptyset$ due to \Cref{ass:ddall_plant_exo}.

  To prove item (ii), note that $\bbssoutp$ is a $\bboutpdim-$dimensional vector function whose scalar components are linear combinations of the $\bbexostatedim$ linearly independent functions generated by the exosystem; hence, for any fixed $\bbexostate_0\in\bbexoinitset$, the ensuing $\bbssoutp$ is uniquely associated to a set of $\bboutpdim \bbexostatedim$ scalars. Consequently, if \Cref{alg:ddall_algo1} is able to determine a matrix $\bar{M}$ (possibly different from the \emph{true} $\bar{M}_{\bboutp,\bbexostate}$ solving \eqref{eq:ddall_Mbar}) which ensures that $\bar{M}\bbexostate$ interpolates $\bbssoutp$ at $\bboutpdim \bbexostatedim$ (or more) points in time, then for the fixed $\bbexostate_0$ the same $\bar{M}$ interpolates $\bbssoutp$ at any time, since it matched all the $\bboutpdim \bbexostatedim$ degrees of freedom in $\bboutpdim$. Since \Cref{alg:ddall_algo1} is based on satisfying \eqref{eq:ddall_menomista2} at $K\geq \bbstatedim+1+\bboutpdim \bbexostatedim>\bboutpdim \bbexostatedim$ sample points, the claim in item (ii) is proved once it is clear that \eqref{eq:ddall_menomista2} expresses a matching condition for the relation $\bbssoutp(t)=\bar{M}_{\bboutp,\bbexostate}\bbexostate(t)$ at time $t=k\tau_s$. To cancel out the transient terms in \eqref{eq:ddall_mista}, a linear combination of responses has been performed in \eqref{eq:ddall_menomista}, hence \eqref{eq:ddall_menomista2} describes the steady-state response due to the initial state $\bar{\bbexostate}_0=\rho_{A_d}(\bbexostate_d)\bbexostate_0$ (see \eqref{eq:ddall_menomista}). Note that $\bar{\bbexostate}_0$ excites in $\exodynmat_d$ all and only the natural modes excited by $\bbexostate_0$ in $\exodynmat_d$, which implies that the coefficients of such modes are exactly estimated.

  To prove item (iii), denote by $T_\bbexostate$ the block depending on $\bbexostate$ in \eqref{eq:ddall_Toeplitz}. Note that $T_\bbexostate$ is full column rank if and only if
  \begin{align}
    \label{eq:ddall_exciting0}
	  \rank\left(\bbm{\bbexostate(0) & \bbexostate(\tau_s) & \cdots & \bbexostate((K-1)\tau_s)}\right) = \bbexostatedim.
  \end{align}
  Since $K>\bbexostatedim$ and each element $\bbexostate(\tau_s)$ can be expressed as $\bbexostate(\tau_s)=\exodynmat_d^{k} \bbexostate_0$, Cayley-Hamilton's theorem implies that \eqref{eq:ddall_exciting0} is equivalent to:
  \begin{align}
    \label{eq:ddall_exciting1}
	  \rank\left(\bbm{\bbexostate_0 & \exodynmat_d \bbexostate_0 & \cdots & \exodynmat_d^{\bbexostatedim-1} \bbexostate_0}\right)=\bbexostatedim.
  \end{align}
  The condition in \eqref{eq:ddall_exciting1} has the form of a reachability matrix test for the pair $(\exodynmat_d,\bbexostate_0)$, which is equivalent to $\bbexostate_0$ exciting all natural modes of $\exodynmat_d$ if $\exodynmat_d$ has all distinct eigenvalues. By \Cref{ass:ddall_nonpatho}, $\exodynmat_d$ has all distinct eigenvalues since, by \Cref{ass:ddall_plant_exo}, $\exodynmat$ has all distinct eigenvalues; this implies, in turn, that $\exodynmat$ and $\exodynmat_d$ have the same eigenvectors, so that $\bbexostate_0$ excites all natural modes of $\exodynmat_d$ if and only if it excites all natural modes of $\exodynmat$. Hence, $T_\bbexostate$ is full column rank by the condition in item (iii).

  Let  $T_{\bboutp}$ denote the block depending on $\bboutp$ in \eqref{eq:ddall_Toeplitz}, and by $T_{\bboutp}^h$ the submatrix containing its rightmost $h$ columns. Since the columns of $T_\bbexostate$ are a linearly independent set, the algorithm starts to consider matrices $[T_{\bboutp}^h \;\; T_\bbexostate]$ increasing $h$ until for $h=\nu$, it happens that $[T_{\bboutp}^{\nu-1} \;\; T_\bbexostate]$ is still full column rank, whereas $[T_{\bboutp}^{\nu} \;\; T_\bbexostate]$ has a non-trivial kernel, which has necessarily dimension one since $[T_{\bboutp}^{\nu} \;\; T_\bbexostate]$ has exactly one column more than $[T_{\bboutp}^{\nu-1} \;\; T_\bbexostate]$. Thus, there is a unique solution $v$ to \eqref{eq:ddall_aM_hat} with the structure in \eqref{eq:ddall_structure_v} and minimal value of $\nu$. Since a solution \eqref{eq:ddall_Minv} is found in item (i) on the basis of \eqref{eq:ddall_structure_v} and since the latter is shown to be uniquely defined, under the stated hypotheses the unique solution found by \Cref{alg:ddall_algo1} must result in the same $\bar{M}$ arising from \eqref{eq:ddall_Mbar}.
\end{proof}

\begin{remark}
  Although \Cref{alg:ddall_algo1} is provided in this context with the specific purpose of estimating $\bar{M}_{\bboutp,\bbexostate}$ to compute the dynamic allocator \eqref{eq:ddall_control_all}, it is a general-purpose algorithm for the estimation of the pseudo-steady-state output map of a linear system such as, \emph{e.g.}, system \eqref{eq:moments_my_sys}-\eqref{eq:moments_my_sys_exosys} in \Cref{app:moments}. Thus, its interest falls within the broader context of online data-driven moment estimation for linear systems, as do the statements of \Cref{prop:ddall_est_mom} and its proof requiring only an adapted notation.\closeremark
\end{remark}

\Cref{alg:ddall_algo1} provides a procedure to compute in finite time the information summarized in the item \textbf{K1}, for any $\mu\in\mathcal{M}$. Conversely, computing \textbf{K2} entails estimating the moments $W_{\bboutp\bbaux}(0), W_{\bboutp\bbaux}(j \omega_i)$, $\omega_i\in\spec(\exodynmat)$ , $i = 1, ..., \bar{n}_e$. In this case, however, the measurements of $\bboutp(t)$ are not influenced only by $\bbaux(t)$ but also by $\bbexostate(t)$ (since the exosystem cannot be disconnected at any given time). Thanks to the computation of \textbf{K1}, $\bar{M}_{\bboutp,\bbexostate}$ is now available. The next statement follows trivially from the previous results.
\begin{lemma}
  \label{lem:ddall_algoK2}
  If \textbf{K1} is available, then \Cref{alg:ddall_algoK2} computes an estimate of \textbf{K2}.\closestatement
\end{lemma}

\begin{algorithm}[h!]
  \caption{Data-driven computation of \textbf{K2}.}
  \label{alg:ddall_algoK2}
  \begin{algorithmic}[1]
    \Require $\bar{M}_{\bboutp,\bbexostate}$, $\Omega$, $\Lambda$, measurements of $\bboutp(t)$.
    \Ensure $W_{\bboutp\bbaux}(0), W_{\bboutp\bbaux}(j \omega_i)$, $\omega_i\in\spec(\exodynmat)$, $i = 1, ..., \bar{n}_e$.

    \State Define
    \Statex
      \begin{align}
        \bboutp^{\delta}(t) = \bboutp(t) - \bar{M}_{\bboutp,\bbexostate}\bbexostate(t), \quad t\in\R_{\geq 0}.
      \end{align}

    \State Define the \emph{virtual dynamics}
    \Statex
      \begin{subequations}
        \label{eq:ddall_virtual_exo}
        \begin{align}
          \dot{\zeta}(t) &= \Omega\zeta(t), & t\in\R_{\geq 0}, \\
          \psi(t) &= \Lambda\zeta(t), & t\in\R_{\geq 0},
        \end{align}
      \end{subequations}
    \Statex with $\zeta(0)=\Lambda'$.

    \For{$i=1,\dotsc,\bbauxdim$}
      \State Apply $\bbaux(t)=e_i\psi(t)$, $t\in\R_{\geq 0}$, and measure $\bboutp^{\delta}(t)$.
      \State Apply \Cref{alg:ddall_algo1} to the measurements of $(\zeta(t),\bboutp^{\delta}(t))$, yielding
      \Statex
        \begin{align}
          \bar{M}_{i} = \bbm{\bar{M}_{i,0} & \bar{M}_{i,1} & \ldots & \bar{M}_{i, 2 \bar{n}_e}},
        \end{align}
      \Statex \hspace{.5cm}with $\bar{M}_{i,h} \in \mathbb{R}^{\bboutpdim}$, $h=0,\dotsc,2\bar{n}_e$.
      \State The $i$-th column of $W_{\bboutp\bbaux}(0)$ is $\bar{M}_{i,0}$.
      \State The $i$-th column of $W_{\bboutp\bbaux}(j\omega_h)$ is $\bar{M}_{i,2h-1} + j\bar{M}_{i,h}$, $h=1,\dotsc,\bar{n}_e$.
    \EndFor
  \end{algorithmic}
\end{algorithm}

Finally, the key step to determine \textbf{K3} is to find a polynomial annihilator $N_a(s)$ such that its columns are a minimal polynomial base of the null space of $W_{\bbouto\bbaux}(s)$. It is then transformed into a rational annihilator $W_a(s)$ by adding a Hurwitz denominator to make it proper. Since the numerators in the elements of $W_{\bbouto\bbaux}(s)$ have degree at most $\bbstatedim$, then, according to \cite[Prop. 1]{zuniga_anaya_improved_2009}, $N_a(s)$ can be chosen with elements of degree not larger than $\bbstatedim\bboutodim$ exactly by selecting the columns of $N_a(s)$ as a minimal polynomial base of the null space of $W_{\bbouto\bbaux}(s)$ \cite[Main Thm.]{forney_jr_minimal_1975}.

Define
\begin{subequations}
  \label{eq:ddall_ncnr}
  \begin{align}
    n_c &= \bbstatedim\bboutodim + 1, \\
    n_r &= \min\{v\in\N : 2v\bboutodim \geq \bbauxdim n_c\}.
  \end{align}
\end{subequations}
For a set of frequencies $\{\bar{\omega}_1, ..., \bar{\omega}_{n_r}\}$, define the block matrix $H$ with the $(k,h)$-th block of size $2\bboutodim\times\bbauxdim$ as
\begin{equation}
  \label{eq:ddall_H_hk}
  H_{[k,h]} =
		\bbm{
		\re\left((j\bar{\omega}_k)^{h-1}W_{\bbouto\bbaux}(j\bar{\omega}_k)\right) \\
		\im\left((j\bar{\omega}_k)^{h-1}W_{\bbouto\bbaux}(j\bar{\omega}_k)\right)
		}, \quad k=1,\dotsc,n_r, \; h=1,\dotsc,n_c,
\end{equation}
where $\rank(W_{\bbouto\bbaux}(j\bar{\omega}_k)) = \bboutodim$ provided that $s=j\bar{\omega}_k$ is not a pole or a zero of the transfer matrix $W_{\bbouto\bbaux}(s)$.

The computation of a polynomial annihilator for $W_{\bbouto\bbaux}(s)$ can be translated into the computation of the null space of the real matrix $H$. The key issue is that, while every polynomial vector in a minimal basis of the null space of $W_{\bbouto\bbaux}(s)$ is associated to a single real vector in the kernel of $H$, the opposite transformation maps a basis of $\ker(H)$ to a linearly dependent set of polynomial vectors in the null space of $W_{\bbouto\bbaux}(s)$. This problem is avoided by computing a \emph{minimal coefficient basis} for the null space of $H$, whose columns, when translated back into their polynomial form, yield a \emph{minimal polynomial basis} for the null space of $W_{\bbouto\bbaux}(s)$ with dimension $\rho=\bbauxdim-\bboutodim$. The following \Cref{alg:ddall_mincoeffbasis} is adapted from \cite{zuniga_anaya_improved_2009}, where several possible optimized computations are also proposed. A minimal \emph{polynomial} basis matrix of order $h-1$ is computed from a minimal \emph{coefficient} basis matrix $N\in\R^{\bbauxdim h\times\rho}$ as
\begin{equation}
  \bbm{I_\bbauxdim & s I_\bbauxdim & \cdots & s^{h-1}I_\bbauxdim}N.
\end{equation}

\begin{algorithm}[h!]
  \caption{Computation of a minimal coefficients basis.}
  \label{alg:ddall_mincoeffbasis}
  \begin{algorithmic}[1]
    \Require $H = \bbm{H_{[:,1]} & H_{[:,2]} & \ldots}$.
    \Ensure A minimal coefficient basis $N$ for $H$.

    \State $\bar{\Sigma} \gets H_{[:,1]}$
    \State $k_1 \gets \dim(\ker(\bar{\Sigma}))$
    \State $\bar{p} \gets \emptyset$
    \State $h \gets 1$
    \If{$k_1=0$}
        \State $X_2 \gets 0_{1\times(2\bbauxdim)}$
        \State $h \gets 2$
        \State \textbf{goto} 15
    \ElsIf{$0<k_1<\rho$}
        \State $X_1 \gets 0_{1\times\bbauxdim}$
        \State \textbf{goto} 22
    \ElsIf{$k_1=\rho$}
        \State \textbf{goto} 29
    \EndIf

    \State $\bar{\Sigma}\gets\bbm{H_{[:,1:h]}' \\ X_h'}$
    \State $k_h \gets \dim(\ker(\bar{\Sigma}))$
    \If{$k_h=\rho$}
        \State \textbf{goto} 29
    \Else
        \State \textbf{goto} 22
    \EndIf

    \State $\bar{Q}$, $\bar{R}$ $\gets$ QR decomposition of $\bar{\Sigma}'$.
    \Statex \Comment{$\bar{\Sigma}'=\bar{Q}\bar{R}$ with $\bar{Q}$ orthogonal and $\bar{R}$ upper-triangular.}
    \State $\bar{L}$ $\gets$ The last $(p-\rank(\bar{R}))$ rows of $\bar{Q}'$.
    \State $\bar{p}\gets\bar{p}\cup\bar{p}_h$, where $\bar{p}_h$ is the set of pivot indices in a reduced row echelon form of $\bar{L}$.
    \State $\bar{X}\gets$ The rows of $I_\bbauxdim$ with indices in $\bar{p}$.
    \State $X_{h+1} \gets \diag(X_h, \bar{X})$
    \State $h \gets h+1$
    \State \textbf{goto} 15

    \State $N\in\R^{\bbauxdim h\times\rho}\gets$ Any basis matrix of $\ker(\bar{\Sigma})$.
  \end{algorithmic}
\end{algorithm}

\begin{proposition}
  \label{prop:ddall_annih_computation}
  Recall \eqref{eq:ddall_ncnr}. Consider the transfer matrix $W_{\bbouto\bbaux}(s)\in\C^{\bboutodim\times\bbauxdim}$ and an arbitrary collection of frequencies $\{\bar{\omega}_1,\dotsc,\bar{\omega}_\bbstatedim\}$ such that $\rank(W_{\bbouto\bbaux}(j\bar{\omega}_i))=\bboutodim$, $i=1,\dotsc,\bbstatedim$. Let the matrix $H\in\R^{\bboutodim n_r \times \bbauxdim n_c}$ be defined as in \eqref{eq:ddall_H_hk}. Let $N\in\R^{\bbauxdim h\times(\bbauxdim - \bboutodim)}$ be a \emph{minimal coefficient basis} matrix computed by \Cref{alg:ddall_mincoeffbasis} and partitioned as
  \begin{equation}
    N = \bbm{N_0 \\ N_1 \\ \vdots \\ N_{h-1}}, \quad N_i\in\R^{\bbauxdim\times(\bbauxdim - \bboutodim)}, \; i=0,\dotsc,h-1.
  \end{equation}
  Then, the matrix function
  \begin{equation}
    \label{eq:ddall_Na}
    N_a(s) = N_0 + sN_1 + \ldots + s^{h-1}N_{h-1}
  \end{equation}
  is a polynomial annihilator of $W_{\bbouto\bbaux}(s)$ whose columns are a minimal polynomial basis for the null space of $W_{\bbouto\bbaux}(s)$.\closestatement
\end{proposition}
\begin{proof}
  Since each element in the polynomial matrix $\det(sI-A)W_{\bbouto\bbaux}(s)$ has degree not larger than $\bbstatedim$ and $\rank(\det(sI-A)W_{\bbouto\bbaux}(s))=\bbouto$ by \Cref{ass:ddall_plant_exo}, it follows from \cite[Prop. 1]{zuniga_anaya_improved_2009} that a polynomial annihilator $N_a(s)$ with entries having maximum degree not larger than $\bbstatedim\bboutodim$ exists. Then, the equation $W_{\bbouto\bbaux}(s)N_a(s) = 0$ has a solution of the form
  \begin{equation}
    N_a(s) = N_0 + sN_1 + \dotsc + s^{\bar{n}}N_{\bar{n}}, \quad \bar{n} < n_c.
  \end{equation}
	Let
  \begin{equation}
    N_{\bar{n}} = \bbm{N_0 \\ N_1 \\ \vdots \\ N_{\bar{n}}}
  \end{equation}
  so that
  \begin{equation}
    N_a(s) = \bbm{I_\bbauxdim & sI_\bbauxdim & \cdots & s^{\bar{n}}I_\bbauxdim}N_{\bar{n}}.
  \end{equation}
  Thus, the annihilator equation to be solved becomes
  \begin{equation}
    \label{eq:ddall_solveme}
    W_{\bbouto\bbaux}(s)N_a(s) = W_{\bbouto\bbaux}(s)\bbm{I_\bbauxdim & sI_\bbauxdim & \dotsc & s^{\bar{n}}I_\bbauxdim}N_{\bar{n}} = 0.
  \end{equation}
  By evaluating \eqref{eq:ddall_solveme} at the frequencies $\{\bar{\omega}_1,\dotsc,\bar{\omega}_\bbstatedim\}$ and imposing that both real and imaginary parts of the left hand side of \eqref{eq:ddall_solveme} are equal to zero, it follows that $N_{\bar{n}}$ lies in the null space of the matrix $H_{[:,1:\bar{n}]}$ whose blocks are defined in \eqref{eq:ddall_H_hk}. Finally, arguments similar to those in \cite{zuniga_anaya_improved_2009}, \Cref{alg:ddall_mincoeffbasis} computes the minimal $\bar{n}=h$ and a suitable $N_{\bar{n}}=N$, yielding the desired result.
\end{proof}

\begin{remark}
  \label{rem:ddall_freqschoice}
  The requirement of selecting frequencies $\{\bar{\omega}_i\}_{i=1}^\bbstatedim$ with a full-rank property, namely	such that $\rank(W_{\bbouto\bbaux}(j\bar{\omega}_i)) = \bboutodim$, $i = 1,\dotsc,\bbstatedim$,	originates from the necessity to rule out pathological cases that may	arise in control systems such as those modeled by \eqref{eq:ddall_sys}. More precisely, suppose that \eqref{eq:ddall_sys} embeds a feedback control loop (recall the linear case of \Cref{sec:all_design_ssinv}), which has been designed by incorporating an internal model of the exosystem, as is customary in output regulation problems. In that case, choosing frequencies	that belong to the internal model yields null rows in $H$, making the polynomial annihilator's computation unfeasible. It is worth observing that, in the uncertain setting, a random selection of such frequencies \emph{generally} ensures the required full-rank property.\closeremark
\end{remark}

\begin{remark}
  \label{rem:ddall_toeplitz}
  \Cref{alg:ddall_mincoeffbasis} can be applied to find a minimal coefficient basis	for a polynomial annihilator of the polynomial matrix	$A(s)=s^{\bar{n}}A_{\bar{n}}+\cdots+sA_1+A_0$ where $A_i\in\R^{\bboutodim\times\bbauxdim}$ and $\rank(A(s))=\bboutodim$. In such a case, instead of forming matrix $H$ with blocks described by	\eqref{eq:ddall_H_hk}, it is possible to use a block Toeplitz matrix $T$ whose $(k,h)-$th block is $T_{[k,h]}=A_{k-h}$ if $\,0\leq k-h \leq \bar{n}$, and	$T_{[k,h]}=0$ otherwise. This applies to, \emph{e.g.}, the case of $\mu=0$, in which the polynomial matrix $N_{\bbouto\bbaux}(s)=\det(sI-A)W_{\bbouto\bbaux}(s)$ is known.\closeremark
\end{remark}

Recalling the definition of matrix $H$ in \eqref{eq:ddall_H_hk}, it appears evident that \Cref{prop:ddall_annih_computation} is stated with the ideal premise of a perfect knowledge of the transfer matrix $W_{\bbouto\bbaux}(s)$ evaluated at the frequencies $\{\bar{\omega}_1,\dotsc,\bar{\omega}_n\}$; such a knowledge is not available for a generic $\mu\neq 0$. Therefore, to extend the previous results to the case of a generic $\mu\in\mathcal{M}$, it is necessary to perform a data-driven estimation of the moments $W_{\bbouto\bbaux}(j\bar{\omega}_i)$, $i=1,\dotsc,\bbstatedim$. The proposed procedure is illustrated in \Cref{alg:ddall_computeH}.

\begin{algorithm}[h!]
  \caption{Computation of $H$ in \eqref{eq:ddall_H_hk}.}
  \label{alg:ddall_computeH}
  \begin{algorithmic}[1]
    \Require $\{\bar{\omega}_i\}_{i=1}^{\bbstatedim}$, $\tau_s$, $K\in\{a\in\N : a > K > \bbstatedim+\bbexostatedim+1\}$.
    \Ensure $H$.

    \State Define the \emph{virtual dynamics}
    \Statex
      \begin{subequations}
        \label{eq:ddall_OmegaLambdaEv_bar}
        \begin{align}
          \bar{\Omega} &= \diag\left(\bbm{0 & \bar{\omega}_{1} \\ -\bar{\omega}_1 & 0}, \dotsc, \bbm{0 & \bar{\omega}_n \\ -\bar{\omega}_n & 0}\right),\label{eq:ddall_Omega_bar} \\
          \bar{\Lambda} &= \bbmat{cc:c:cc}{c}{1 & 0 & \cdots & 1 & 0},\label{eq:Lambda_bar} \\
          \dot{\zeta}(t) &= \bar{\Omega}\zeta(t), & t\in\R_{\geq 0},\label{eq:ddall_virtual_exo2} \\
          \psi(t) &= \bar{\Lambda}\zeta(t), & t\in\R_{\geq 0},\label{eq:ddall_virtual_exo2_out}
        \end{align}
      \end{subequations}
    \Statex with $\zeta(0)=\bar{\Lambda}'$.

    \State $\bbaux(t) \gets 0$, $t\in\R_{\geq 0}$
    \State Collect samples $\mathcal{W}_d = \{\bbexostate(k\tau_s)\}_{k=0}^{\bbstatedim+K-1}$, $\mathcal{H}_d=\{\bbouto(k\tau_s)\}_{k=0}^{\bbstatedim+K-1}$.

    \State Apply \Cref{alg:ddall_algo1} to $(\mathcal{W}_d,\mathcal{H}_d)$ to obtain $\bar{M}_{\bbouto,\bbexostate}$.
    \State $\bbouto^{\delta}(t) \gets \bbouto(t) - \bar{M}_{\bbouto,\bbexostate}\bbexostate(t)$

    \For{$i=1,\dotsc,\bbauxdim$}
        \State $\bbaux(t) \gets e_i\psi(t)$, $t\in\R_{\geq 0}$
        \Statex \Comment{Connect \eqref{eq:ddall_OmegaLambdaEv_bar} to the $i$-th input channel.}
        \State Collect samples $\mathcal{Z}_d = \{\zeta(k\tau_s)\}_{k=0}^{\bbstatedim+K-1}$, $\bar{\mathcal{H}}_d = \{\bbouto^{\delta}(k\tau_s)\}_{k=0}^{\bbstatedim+K-1}$.
        \State Apply \Cref{alg:ddall_algo1} to $(\mathcal{Z}_d, \bar{\mathcal{H}}_d)$ and obtain $\bar{M}_{\bbouto,\zeta}^i$ partitioned as
        \Statex
          \begin{equation}
            \label{eq:ddall_Mbar_zeta}
            \bar{M}^{i}_{\bbouto,\zeta} = \bbm{
              \re\left(W^{i}_{\bbouto\bbaux}(j\bar{\omega}_1)\right)' \\
              \im\left(W^{i}_{\bbouto\bbaux}(j\bar{\omega}_1)\right)' \\
              \vdots                                                  \\
              \re\left(W^{i}_{\bbouto\bbaux}(j\bar{\omega}_n)\right)' \\
              \im\left(W^{i}_{\bbouto\bbaux}(j\bar{\omega}_n)\right)'
            }'.
          \end{equation}
    \EndFor

    \State Rearrange the columns of $\{\bar{M}^{i}_{\bbouto,\zeta}\}_{i=1}^\bbauxdim$ to obtain $H$ as in \eqref{eq:ddall_H_hk}.
  \end{algorithmic}
\end{algorithm}

The following result summarizes all the previous claims, and proves that the proposed control allocator is determined for any $\mu\in\mathcal{M}$.

\begin{theorem}
  \label{thm:ddall_main}
  Consider the system \eqref{eq:ddall_sys}-\eqref{eq:ddall_exo} with $\mu\in\mathcal{M}$. Suppose that \Cref{ass:bbiss,ass:ddall_plant_exo,ass:ddall_redundancy,ass:ddall_nonpatho} hold. Then, the dynamic control allocator \eqref{eq:ddall_control_all} interconnected to \eqref{eq:ddall_sys}-\eqref{eq:ddall_exo} as in \eqref{eq:ddall_all_interc} solves \Cref{prb:ddall_main_per} for any $\gamma\in\R_{>0}$.\closestatement
\end{theorem}
\begin{proof}
  The validity of \Cref{thm:ddall_main} follows from the combination of all the previous results. Indeed, the full design procedure for any $\mu\in\mathcal{M}$ can now be illustrated, providing a constructive proof.
  Consider the data \textbf{K1}, \textbf{K2}, \textbf{K3} in \Cref{def:ddall_data}.
  If $\mu=0$, \emph{i.e.}, \eqref{eq:ddall_sys} is fully known, then \Cref{rem:ddall_nominalK123} provides a way to compute all the items.
  Conversely, if $\mu\neq 0$, it is necessary to perform an \emph{online estimation} of all the data items. The procedure is summarized as follows.
  \begin{enumerate}
    \item \textbf{K1} is determined using \Cref{alg:ddall_algo1}, whose validity is proven by \Cref{prop:ddall_est_mom}.
    \item \textbf{K2} is given by \Cref{alg:ddall_algoK2}, whose correctness is guaranteed by \Cref{lem:ddall_algoK2}.
    \item \textbf{K3}, \emph{i.e.}, the polynomial annihilator of \eqref{eq:ddall_sys} with respect to the regulated output, is determined by first computing the matrix $H$ in \eqref{eq:ddall_H_hk} using \Cref{alg:ddall_computeH}, then using it in \Cref{alg:ddall_mincoeffbasis} to obtain a minimal polynomial basis for the null space of $W_{\bbouto\bbaux}(s)$, as stated in \Cref{prop:ddall_annih_computation}.
  \end{enumerate}
  Once the data \textbf{K1}, \textbf{K2}, and \textbf{K3} have been estimated, it is possible to build the control allocator \eqref{eq:ddall_control_all}, which solves \Cref{prb:ddall_main_per} for any $\gamma\in\R_{>0}$ as stated in \Cref{lem:ddall_nominal}.
\end{proof}

  \begin{equation}
    \label{eq:ddall_example_AC2}
    A_{\Theta_2} = \left[\begin{array}{cc}A_p(0) & B_p(0)\Xi C_{\Theta_1} \\ 0 & A_{\Theta_1}\end{array}\right], \quad B_{\Theta_2} = \left[\begin{array}{c} 0 \\ I \end{array}\right], \quad C_{\Theta_2} = \left[\begin{array}{cc} C_p(0) & 0 \end{array}\right].
  \end{equation}
  The gain matrices
  \begin{subequations}
    \label{eq:ddall_example_KC2}
    \begin{align}
      L_{\Theta_2} & = \bbm{10.85 & -3.51 & 12.25   & 1.40  & -0.17}', \\
      K_{\Theta_2} & =
      \bbm{-8.09        & -4.41        & -3.40 & -8.07   & -0.92            \\ -0.41 &
      -0.89             & -0.44        & -0.92 & -1.41},
    \end{align}
  \end{subequations}
  have been obtained with an LQR approach. Finally, the regulated and performance outputs are defined as in \eqref{eq:ddall_sys_z0} and \eqref{eq:ddall_sys_z}, respectively, with
  \begin{equation}
    C_1(\mu) = \left[\begin{array}{cc} C_p(\mu) & 0 \end{array} \right], \quad C_2(\mu) = \left[\begin{array}{ccc} 0 & C_{c}\end{array}\right], \quad C_c = \left[\begin{array}{cc} C_{\Theta_1} & 0 \end{array}\right],
  \end{equation}
  and $D_{21} = 0$, $D_{12} = Q$, $D_{21} = I$, $D_{22} = 0$.

  Denote by $y_c(t)\in\R^2$ the output vector of the full regulator and by $u(t)\in\R^2$ the input vector of the plant under control. The interconnection of the plant, the regulator, and the allocator to be designed is given by
  \begin{equation}
    u(t) = y_c(t) + \varsigma(t), \quad t\in\R_{\geq 0}.
  \end{equation}
  A polynomial annihilator of the nominal transfer matrix $W_{\bbouto\bbaux}(s)$ is
  \begin{equation}
    N_a(s) = \bbm{-s^2-s-3 \\ -3s-3}, \quad s\in\C,
  \end{equation}
  hence, its state-space description is given by
  \begin{equation}
    \label{eq:ddall_example_control_all_ya}
    A_a =  \bbm{-3 & -3 & -1 \\ 1 & 0 & 0 \\ 0 & 1 & 0}, \quad B_a=\bbm{1 \\ 0 \\ 0},	\quad	C_a = \bbm{-1 & -1 & -3 \\ 0 & -3 & -3}.
  \end{equation}
  The optimization unit is designed as in \eqref{eq:ddall_control_all} with $R = I$ for multiple values of $\gamma$.
