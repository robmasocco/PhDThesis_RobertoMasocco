To investigate this, consider the continuous-time, linear time-invariant plant $\plant$ given by
\begin{subequations}
  \label{eq:sdall_ctsys}
  \begin{align}
    \dot{\plantstate}(t) &= \plantdynmat \plantstate(t) + \plantctrmat \plantinput(t), & t\in\R_{\geq 0},\\
    \plantoutput(t) &= \plantoutmat \plantstate(t), & t\in\R_{\geq 0},
  \end{align}
\end{subequations}
where $\plantstate(t)\in\R^{\plantstatedim}$ is the plant state vector, $\plantinput(t)\in\R^{\plantinputdim}$ is the plant input vector, and $\plantoutput(t)\in\R^{\plantoutputdim}$ is the plant output vector at time $t$, and $\plantdynmat\in\R^{\plantstatedim\times\plantstatedim}$, $\plantctrmat\in\R^{\plantstatedim\times\plantinputdim}$, and $\plantoutmat\in\R^{\plantoutputdim\times\plantstatedim}$ are constant matrices. Let $\tau_s\in\R{\geq 0}$ be the sampling time. The plant \eqref{eq:sdall_ctsys} is part of a feedback control scheme comprising a sampled-data controller $\ctrl$ described by the following discrete-time, linear time-invariant state-space model:
\begin{subequations}
  \label{eq:sdall_dtcontroller}
  \begin{align}
      \ctrlstate_{k+1} &= \ctrldynmat\ctrlstate_k + \ctrlctrmat\plantoutput_k + \ctrlrefmat\bbref_k, & k\in\Z_{\geq 0},\\
      \ctrlouput_k &= \ctrloutmat\ctrlstate_k + \ctrldirgain\plantoutput_k + \ctrlrefgain\bbref_k, & k\in\Z_{\geq 0},
  \end{align}
\end{subequations}
where $\ctrlstate_k\in\R^{\ctrlstatedim}$ is the controller state vector, and $\ctrlouput_k\in\R^{\ctrloutputdim}$ is the controller output vector at time $k$. The matrices $\ctrldynmat\in\R^{\ctrlstatedim\times\ctrlstatedim}$, $\ctrlctrmat\in\R^{\ctrlstatedim\times\plantoutputdim}$, $\ctrlrefmat\in\R^{\ctrlstatedim\times\bbrefdim}$, $\ctrloutmat\in\R^{\ctrloutputdim\times\ctrlstatedim}$, $\ctrldirgain\in\R^{\ctrloutputdim\times\plantoutputdim}$, and $\ctrlrefgain\in\R^{\ctrloutputdim\times\bbrefdim}$ are constant. $r(t)\in\R^{\bbrefdim}$ is the reference input at time $t$, generated by an exosystem of the form \eqref{eq:bbexosys}. Let $\dplantdynmat\in\R^{\plantstatedim\times\plantstatedim}$, $\dplantctrmat\in\R^{\plantstatedim\times\plantinputdim}$, and $\plantoutmat\in\R^{\plantoutputdim\times\plantstatedim}$ be constant matrices given by
\begin{subequations}
  \label{eq:sdall_discretization}
  \begin{align}
    \dplantdynmat &= e^{\plantdynmat\tau_s},\\
    \dplantctrmat &= \int_{0}^{\tau_s} e^{\plantdynmat(\tau_s - \sigma)}\plantctrmat\,d\sigma,\\
    \dplantoutmat &= \plantoutmat.
  \end{align}
\end{subequations}
The matrices in \eqref{eq:sdall_discretization} yield the following \emph{discretized} state-space description of the plant $\plant$, equivalent to \eqref{eq:sdall_ctsys} at the sampling times $t=k\tau_s$, $k\in\Z_{\geq 0}$:
\begin{subequations}
  \label{eq:sdall_dtsys}
  \begin{align}
    \plantstate_{k+1} &= \dplantdynmat\plantstate_k + \dplantctrmat\plantinput_k, & k\in\Z_{\geq 0},\\
    \plantoutput_k &= \dplantoutmat\plantstate_k, & k\in\Z_{\geq 0}.
  \end{align}
\end{subequations}
Let $\{0,\omega_1,\dotsc,\omega_h\}$ be natural pulses such that there exist $\bbperiod\in\R_{>0}$ and $\{k_i\}_{i=1}^{h}\subset\N$ such that
\begin{align}
  \label{eq:sdall_exofreqs}
  \omega_i &= k_i\frac{2\pi}{\bbperiod}, & i=1,\dotsc,h.
\end{align}
The characteristics of the control system resulting from the feedback interconnection of \eqref{eq:sdall_ctsys} and \eqref{eq:sdall_dtcontroller}, as well as the properties of the exosystem considered in this scenario, are stated by the following assumptions.
\begin{assumption}
  \label{ass:sdall_controller}
  The pair $(\plantdynmat,\plantctrmat)$ is controllable, and the pair $(\plantoutmat,\plantdynmat)$ is detectable.
  The design of the discrete-time controller \eqref{eq:sdall_dtcontroller} is based on the discrete-time equivalent description \eqref{eq:sdall_dtsys} of the plant. Moreover, the feedback interconnection of \eqref{eq:sdall_ctsys} and \eqref{eq:sdall_dtcontroller} given by $\plantinput=\ctrlouput$ by means of a \emph{zero-order hold} device on the plant input, and a \emph{sample-and-hold} device on the plant output, each with sampling time $\tau_s$, is well-posed and asymptotically stable.\closestatement
\end{assumption}
\begin{assumption}
  \label{ass:sdall_exo}
  The reference input $\bbref$ is generated by the exosystem $\bbexo$ given by
  \begin{subequations}
    \label{eq:sdall_ctexo}
    \begin{align}
      \dot{\bbexostate}(t) &= \exodynmat\bbexostate(t), & t\in\R_{\geq 0},\\
      \bbref(t) &= \exooutmat\bbexostate(t), & t\in\R_{\geq 0},
    \end{align}
  \end{subequations}
  where $\exodynmat\in\R^{\bbexostatedim\times\bbexostatedim}$ is such that
  \begin{subequations}
    \label{eq:sdall_exoblocks}
    \begin{align}
      \exodynmat_0 &= 0,\\
      \exodynmat_i &= \begin{bmatrix}0 & \omega_i \\ -\omega_i & 0\end{bmatrix}, & i=1,\dotsc,h,\\
      \exodynmat &= \diag(\exodynmat_0,\exodynmat_1,\exodynmat_2,\dotsc,\exodynmat_h).
    \end{align}
  \end{subequations}
  The matrix $\exooutmat\in\R^{\bbrefdim\times\bbexostatedim}$ is constant and the pair $(\exooutmat,\exodynmat)$ is observable. Moreover, sampled measurements of the reference input, namely $r_k$ at time $k$, are available by means of a \emph{sample-and-hold} device with the same sampling time $\tau_s$ as in \Cref{ass:sdall_controller}.\closestatement
\end{assumption}
It is worth pointing out that \Cref{ass:sdall_exo} implies that the reference input $r$ is periodic of period $\bbperiod$.
\begin{assumption}
  \label{ass:sdall_periods}
  There exists $K\in\N$ such that $\bbperiod = K\tau_s$.\closestatement
\end{assumption}
Although \Cref{ass:sdall_periods} is instrumental to simplify the following analysis, it can be relaxed in practice as shown later on. Similarly to the continuous-time, linear case presented in \Cref{sec:all_design_ssinv}, the feedback control system given by \eqref{eq:sdall_ctsys}--\eqref{eq:sdall_dtsys} and \eqref{eq:sdall_ctexo}--\eqref{eq:sdall_exoblocks} is a variant of \eqref{eq:bbsys} including discrete-time variables:
\begin{subequations}
  \label{eq:sdall_bb}
  \begin{align}
    \plantinput(t) &= \ctrlouput(k\tau_s) + \bbaux(k\tau_s), & t\in [k\tau_s,(k+1)\tau_s), k\in\Z_{\geq 0},\\
    \bbstate_k &= \begin{bmatrix}\plantstate_k' & \ctrlstate_k'\end{bmatrix}', & k\in\Z_{\geq 0},\\
    \bboutp(t) &= \plantinput(t), & t\in\R_{\geq 0},\\
    \bbouto(t) &= \plantoutput(t), & t\in\R_{\geq 0},
  \end{align}
\end{subequations}
with $\bbstate_k\in\R^{\plantstatedim + \ctrlstatedim}$ and $\bbstatedim = \plantstatedim + \ctrlstatedim$, and $\bbaux_k\in\R^{\bbauxdim}$ is the auxiliary input at time $k$.

Let $R\in\R^{\plantinputdim\times\plantinputdim}$ be a constant matrix such that $R=R'$, $R\succ 0$. The steady-state plant input is denoted by $\plantssinput$. Consider the following cost functional:
\begin{equation}
  \label{eq:sdall_ctcost}
  J = \int_{0}^{\bbperiod} \left\Vert\plantssinput(t)\right\Vert^2_{R}\,dt.
\end{equation}
The dynamic input allocation problem for the feedback control system hereby defined is formulated as follows.
\begin{problem}
  \label{prb:sdall_all}
  Consider the sampled-data feedback control system given by \eqref{eq:sdall_ctsys}--\eqref{eq:sdall_dtsys} interconnected as in \eqref{eq:sdall_bb}, and \eqref{eq:sdall_ctexo}--\eqref{eq:sdall_exoblocks}. Let \Cref{ass:bboveract,ass:sdall_periods,ass:sdall_exo,ass:sdall_controller} hold. Find, if any, a \emph{discrete-time input allocation device} (namely, an \emph{allocator}) such that, when interconnected to the feedback loop according to
  \begin{align}
    \label{eq:sdall_all_interconn}
    \allin &= \bbref, & \allout &= \bbaux,
  \end{align}
  yielding the plant input $\plantinput = \ctrlouput + \allout$, ensures that the following requirements hold.
  \begin{itemize}
    \item[\emph{(I)}] \emph{Sampled output invisibility:} the regulated outputs both in presence and absence of the allocator are such that
      \begin{align}
        \label{eq:sdall_sdinv}
        \bbouto(k\tau_s) &= \bboutoa(k\tau_s), & k\in\Z_{\geq 0}.
      \end{align}
    \item[\emph{(O)}] \emph{Steady-state optimality:} the cost functional \eqref{eq:sdall_ctcost} is brought to its minimum value.
  \end{itemize}
  \closestatement
\end{problem}

\begin{remark}
  \label{rem:sdall_fullinvprb}
  The statement of \Cref{prb:sdall_all} is indeed a sampled-data variant of \Cref{prb:fullinv}. Although the two problems share the steady-state optimality requirement, what differentiates them is the full output invisibility one. From a continuous-time perspective, \emph{i.e.}, the time domain of the plant \eqref{eq:sdall_ctsys}, \Cref{prb:fullinv} requires the effect of the allocator to be invisible in the plant output at all times. Conversely, \Cref{prb:sdall_all} requires the effect of the allocator to be invisible only at the sampling times. Conceptually, this difference is motivated by the fact that the control system is implemented using digital devices that are now modeled with discrete-time dynamics; this fact reasonably leads to the formulation of the full output invisibility requirement given in \Cref{prb:sdall_all}.\closeremark
\end{remark}

This scenario, completely new in the literature to the best of the author's knowledge, poses two questions which are the subject of the following investigation: how the structure of the allocator introduced in \Cref{sec:all_design_fullinv} (see \Cref{fig:fullallocator}) has to be adapted to a sampled-data design, and how the allocator action affects the plant output between the sampling times \cite{galeani_dynamic_2022}.

\begin{example}
  \label{ex:sdall_ctann}
  The necessity of a discrete-time allocator design approach to solve \Cref{prb:sdall_all} is highlighted by the following numerical example. Consider the continuous-time, linear time-invariant plant of the form \eqref{eq:sdall_ctsys} given by
  \begin{align}
    \label{eq:sdall_cex_ctsys}
    \plantdynmat &= \begin{bmatrix}
      -8 & 2 & -9\\
      9 & -12 & 8\\
      5 & -7 & -3
    \end{bmatrix}, & \plantctrmat &= \begin{bmatrix}
      4 & 7\\
      -10 & -6\\
      10 & -7
    \end{bmatrix}, & \plantoutmat &= \begin{bmatrix}
      -7 & -4 & 1
    \end{bmatrix}.
  \end{align}
  An inspection of \eqref{eq:sdall_cex_ctsys} reveals that the plant is asymptotically stable since $\spec(\plantdynmat)\subset\C_{< 0}$, that the pair $(\plantdynmat,\plantctrmat)$ is controllable, and that the pair $(\plantoutmat,\plantdynmat)$ is observable. Let
  \begin{align}
    \label{eq:sdall_cex_sine}
    v(t) &= 5\sin(10t), & t\in\R_{\geq 0},
  \end{align}
  where $v(t)\in\R$ since \eqref{eq:sdall_cex_ctsys} has degree of redundancy $\bbreddeg = 1$. Then, design a \emph{polynomial annihilator} for \eqref{eq:sdall_cex_ctsys} following the procedure described in \Cref{prop:ddall_annih_computation}. Fix the sampling time $\tau_s =$ \SI{0.1}{\second}. Let $u_a$ be the input and $\allout$ be the output of the resulting annihilator; interconnect the plant \eqref{eq:sdall_cex_ctsys} and the annihilator such that
  \begin{align}
    \label{eq:sdall_cex_interconn}
    u_a(t) &= v(t), & t\in\R_{\geq 0},\\
    \plantinput(t) &= \allout(k\tau_s), & t\in [k\tau_s,(k+1)\tau_s), k\in\Z_{\geq 0},
  \end{align}
  \emph{i.e.}, the annihilator output is fed to the plant \eqref{eq:sdall_cex_ctsys} by means of a zero-order hold device with sampling time $\tau_s$. The situation hereby described is depicted in \Cref{fig:sdall_dtann}. The output of \eqref{eq:sdall_cex_ctsys} in presence of its annihilator subject to the sinusoidal input \eqref{eq:sdall_cex_sine} is shown in \Cref{fig:sdall_y_ctzct}.

\Cref{ex:sdall_ctann} suggests that the entire annihilator design approach should be revised in a discrete-time setting when system \eqref{eq:bbsys} involves sampled-data devices. Let $z$ be a variable taking values in $\C$, and $\dplanttf(z)\in\C^{\plantoutputdim\times\plantinputdim}$ be the transfer function of the discretized plant \eqref{eq:sdall_dtsys}. The approach outlined in \cite{cristofaro_output_2014} and further characterized in \Cref{prop:ddall_annih_computation} is extended to the discrete-time case as follows.

\begin{proposition}
  \label{prop:sdall_annih_computation}
  Let $N_a(z)\in\C^{\ctrloutputdim\times\bbreddeg}$ be a polynomial matrix whose columns form a minimal polynomial basis for the null space of $\dplanttf(z)$, and $d_a(z)$ be a monic polynomial with degree larger than that of each element of $N_a(z)$ and whose roots all lie in $\C_{<1}$.
  Define
  \begin{equation}
    \label{eq:sdall_Wa}
    W_a(z) = N_a(z)d_a^{-1}(z).
  \end{equation}
  Then, it holds that
  \begin{align}
    \label{eq:sdall_annih}
    \dplanttf(z)W_a(z) &= 0, & z\in\C\setminus(\poleset(\dplanttf(z))\cup\poleset(W_a(z))).
  \end{align}
  Furthermore, given any minimal realization $(A_a,B_a,C_a,0)$ of $\,W_a(z)$, the system
  \begin{align}
    \label{eq:sdall_annss}
    {x_a}_{k+1} &= A_a {x_a}_k + B_a v_k, & k\in\Z_{\geq 0},\\
    \allout_k &= C_a {x_a}_k, & k\in\Z_{\geq 0},
  \end{align}
  where $v_k\in\R^{m_a}$, is a \emph{discrete-time annihilator} for plant \eqref{eq:sdall_dtsys}.\closestatement
\end{proposition}

\begin{remark}
  \label{rem:sdall_lcf}
  Suppose that $\dplanttf(z)$ admits a left coprime polynomial factorization \cite[§7.8]{chen_linear_2009} such that
  \begin{equation}
    \label{eq:sdall_lcf}
    \dplanttf(z) = \dplantlcfd(z)^{-1}\dplantlcfn(z),
  \end{equation}
  where $\dplantlcfd(z)$, $\dplantlcfn(z)$ are polynomial matrices.
  Then, the polynomial matrix $N_a(z)$ in \Cref{prop:sdall_annih_computation} may be selected among the solutions of
  \begin{equation}
    \label{eq:sdall_lcf_annih}
    \dplantlcfn(z)N_a(z) = 0.
  \end{equation}
  \closeremark
\end{remark}

\begin{remark}
  \label{rem:sdall_sdinv}
  It is worth pointing out that \eqref{eq:sdall_annss} is a discrete-time system designed such that \eqref{eq:sdall_annih} holds. Hence, the discrete-time annihilator given by \eqref{eq:sdall_annss} satisfies the sampled output invisibility requirement in \Cref{prb:sdall_all}. However, the annihilator \eqref{eq:sdall_annss} does not guarantee output invisibility between sampling instants and the effect of the allocator action on the plant output is not characterized.\closeremark
\end{remark}

The proof of \Cref{prop:sdall_annih_computation} follows a reasoning similar to that of \Cref{prop:ddall_annih_computation}, hence it is omitted.
